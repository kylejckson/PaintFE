cmake_minimum_required(VERSION 3.25)
project(paintfe_heic_encoder LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_PLUGIN_LOADING OFF CACHE BOOL "" FORCE)
set(WITH_KVAZAAR ON CACHE BOOL "" FORCE)
foreach(codec LIBDE265 X265 UVG266 VVDEC VVENC X264 OpenH264_DECODER DAV1D AOM_DECODER AOM_ENCODER SvtEnc RAV1E JPEG_DECODER JPEG_ENCODER OpenJPEG_ENCODER OpenJPEG_DECODER FFMPEG_DECODER OPENJPH_ENCODER WEBCODECS LIBSHARPYUV UNCOMPRESSED_CODEC HEADER_COMPRESSION)
  set(WITH_${codec} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(feature WITH_EXAMPLES WITH_GDK_PIXBUF BUILD_TESTING BUILD_DOCUMENTATION BUILD_DEVELOPMENT_TOOLS)
  set(${feature} OFF CACHE BOOL "" FORCE)
endforeach()
add_compile_definitions(KVZ_STATIC_LIB LIBHEIF_STATIC_BUILD)
add_subdirectory("${HEIF_SOURCE}" libheif)
add_executable(paintfe-heic-encoder main.cpp)
target_include_directories(paintfe-heic-encoder PRIVATE "${HEIF_SOURCE}/libheif/api" "${CMAKE_CURRENT_BINARY_DIR}/libheif")
target_link_libraries(paintfe-heic-encoder PRIVATE heif)
